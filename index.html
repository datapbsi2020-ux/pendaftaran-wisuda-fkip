// --- KONFIGURASI --- // Masukkan ID Folder Google Drive Anda const FOLDER_ID = "MASUKKAN_ID_FOLDER_DRIVE_DISINI"; // 1. FUNGSI MENERIMA/MENYIMPAN DATA (Saat mahasiswa daftar) function doPost(e) { const lock = LockService.getScriptLock(); lock.tryLock(10000); try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheets()[0]; let data; try { data = JSON.parse(e.postData.contents); } catch(err) { return responseJSON({ status: "error", message: "Gagal baca JSON" }); } // Buat Folder Mahasiswa const rootFolder = DriveApp.getFolderById(FOLDER_ID); const folderName = `${data.npm} - ${data.nama}`; const folders = rootFolder.getFoldersByName(folderName); let subFolder = folders.hasNext() ? folders.next() : rootFolder.createFolder(folderName); const saveFile = (base64, name) => { if (!base64 || base64.length < 10) return "-"; try { const split = base64.split(','); const type = split[0].split(';')[0].replace('data:', ''); const decoded = Utilities.base64Decode(split[1]); const blob = Utilities.newBlob(decoded, type, name); const file = subFolder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); return file.getUrl(); } catch (err) { return "-"; } }; const urlIjazah = saveFile(data.files.ijazah, `Ijazah_${data.npm}`); const urlKtp = saveFile(data.files.ktp, `KTP_${data.npm}`); const urlBebas = saveFile(data.files.bebas, `BebasBiaya_${data.npm}`); const urlPkkmb = saveFile(data.files.pkkmb, `PKKMB_${data.npm}`); // Masukkan ke Sheet // PASTIKAN URUTAN INI SAMA DENGAN KOLOM DI SPREADSHEET ANDA sheet.appendRow([ new Date(), // 1. Waktu data.gelombang, // 2. Gelombang data.tahun, // 3. Tahun data.nama, // 4. Nama "'" + data.npm, // 5. NPM (Pakai ' agar tidak jadi rumus) data.prodi, // 6. Prodi data.kelas, // 7. Kelas data.jenjang, // 8. Jenjang data.ttl, // 9. TTL data.jk, // 10. Jenis Kelamin "'" + data.nik, // 11. NIK data.ibu, // 12. Nama Ibu data.email, // 13. Email "'" + data.hp, // 14. No HP data.tgl_lulus, // 15. Tgl Lulus data.ipk, // 16. IPK data.yudisium, // 17. Yudisium data.alamat, // 18. Alamat data.judul_skripsi ? data.judul_skripsi : "-", // 19. JUDUL SKRIPSI urlIjazah, // 20. Link Ijazah urlKtp, // 21. Link KTP urlBebas, // 22. Link Bebas Biaya urlPkkmb, // 23. Link PKKMB subFolder.getUrl() // 24. Link Folder Induk Drive ]); return responseJSON({ status: "success", message: "Tersimpan" }); } catch (err) { return responseJSON({ status: "error", message: err.toString() }); } finally { lock.releaseLock(); } } // 2. FUNGSI MEMBACA DATA (Agar muncul di Tabel Admin Web) function doGet(e) { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheets()[0]; const rows = sheet.getDataRange().getValues(); // Perbaikan: Cek jika data kosong (hanya header) if (rows.length <= 1) { return responseJSON({ status: "success", data: [] }); } rows.shift(); // Hapus Header (Baris pertama) const result = rows.map(row => ({ waktu: row[0] ? row[0] : "", nama: row[3] ? row[3].toString() : "-", npm: row[4] ? row[4].toString().replace(/'/g, "") : "-", prodi: row[5] ? row[5].toString() : "-", hp: row[13] ? row[13].toString().replace(/'/g, "") : "-", link_folder: row[23] || row[19] || "" // Ambil link folder Drive (berada di kolom ke-24 / index 23) })).reverse(); // Urutkan dari yang terbaru ke terlama return responseJSON({ status: "success", data: result }); } catch (err) { return responseJSON({ status: "error", message: err.toString() }); } } function responseJSON(content) { return ContentService.createTextOutput(JSON.stringify(content)).setMimeType(ContentService.MimeType.JSON); }
